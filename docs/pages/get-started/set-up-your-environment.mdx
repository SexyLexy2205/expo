// App.js - TradeGuard React Native app for Android and iOS
import React, { useState } from 'react';
import { NavigationContainer } from '@react-navigation/native';
import { createStackNavigator } from '@react-navigation/stack';
import { createBottomTabNavigator } from '@react-navigation/bottom-tabs';
import {
  View,
  Text,
  Button,
  TextInput,
  FlatList,
  StyleSheet,
  Alert,
  Image,
  Platform,
} from 'react-native';
import AsyncStorage from '@react-native-async-storage/async-storage';
import * as ImagePicker from 'react-native-image-picker';
import { Picker } from '@react-native-picker/picker';

// Stack and Tab Navigators
const Stack = createStackNavigator();
const Tab = createBottomTabNavigator();

// Sample data (replace with backend)
let users = [];
let jobs = [];
let quotes = [];
let reviews = [];

// Homepage
const HomeScreen = ({ navigation }) => (
  <View style={styles.container}>
    <Text style={styles.title}>Welcome to TradeGuard</Text>
    <Image
      source={{ uri: 'https://via.placeholder.com/150' }} // Replace with your logo
      style={styles.logo}
    />
    <Button title="Sign Up" onPress={() => navigation.navigate('SignUp')} />
    <Button title="Login" onPress={() => navigation.navigate('Login')} />
  </View>
);

// Sign Up Screen
const SignUpScreen = ({ navigation }) => {
  const [email, setEmail] = useState('');
  const [password, setPassword] = useState('');
  const [role, setRole] = useState('Customer');
  const [name, setName] = useState('');

  const signUp = async () => {
    if (!email || !password || !name) {
      Alert.alert('Error', 'Fill out all fields, you lazy fuck!');
      return;
    }
    const user = { id: Date.now().toString(), email, password, role, name };
    users.push(user);
    await AsyncStorage.setItem('users', JSON.stringify(users));
    Alert.alert('Success', `Signed up as ${role}!`);
    navigation.navigate(role === 'Customer' ? 'CustomerTabs' : 'TradieTabs');
  };

  return (
    <View style={styles.container}>
      <Text style={styles.title}>Sign Up</Text>
      <TextInput style={styles.input} placeholder="Name" value={name} onChangeText={setName} />
      <TextInput style={styles.input} placeholder="Email" value={email} onChangeText={setEmail} />
      <TextInput
        style={styles.input}
        placeholder="Password"
        value={password}
        onChangeText={setPassword}
        secureTextEntry
      />
      <Picker
        selectedValue={role}
        style={styles.picker}
        onValueChange={(itemValue) => setRole(itemValue)}
      >
        <Picker.Item label="Customer" value="Customer" />
        <Picker.Item label="Tradie" value="Tradie" />
      </Picker>
      <Button title="Sign Up" onPress={signUp} />
    </View>
  );
};

// Login Screen
const LoginScreen = ({ navigation }) => {
  const [email, setEmail] = useState('');
  const [password, setPassword] = useState('');

  const login = async () => {
    const user = users.find((u) => u.email === email && u.password === password);
    if (!user) {
      Alert.alert('Error', 'Wrong credentials, you dumb bastard!');
      return;
    }
    Alert.alert('Success', `Logged in as ${user.role}!`);
    navigation.navigate(user.role === 'Customer' ? 'CustomerTabs' : 'TradieTabs');
  };

  return (
    <View style={styles.container}>
      <Text style={styles.title}>Login</Text>
      <TextInput style={styles.input} placeholder="Email" value={email} onChangeText={setEmail} />
      <TextInput
        style={styles.input}
        placeholder="Password"
        value={password}
        onChangeText={setPassword}
        secureTextEntry
      />
      <Button title="Login" onPress={login} />
    </View>
  );
};

// Customer: Job Posting Screen
const CustomerJobScreen = () => {
  const [name, setName] = useState('');
  const [phone, setPhone] = useState('');
  const [description, setDescription] = useState('');
  const [photos, setPhotos] = useState([]);
  const [address, setAddress] = useState('');

  const pickPhotos = () => {
    ImagePicker.launchImageLibrary({ selectionLimit: 5 }, (response) => {
      if (response.assets) {
        setPhotos(response.assets.map((asset) => asset.uri));
      }
    });
  };

  const postJob = async () => {
    if (!name || !phone || !description || !address || photos.length < 3) {
      Alert.alert('Error', 'Fill out all fields and upload 3–5 photos, you slacker!');
      return;
    }
    const job = {
      id: Date.now().toString(),
      name,
      phone,
      description,
      photos,
      address,
      status: 'Open',
      postedAt: Date.now(),
    };
    jobs.push(job);
    await AsyncStorage.setItem('jobs', JSON.stringify(jobs));
    setTimeout(async () => {
      if (!quotes.some((q) => q.jobId === job.id)) {
        Alert.alert('Notification', `Email sent to tradies near ${address}!`);
      }
    }, 24 * 60 * 60 * 1000); // 24 hours
    Alert.alert('Success', 'Job posted! Escrow payment setup required.');
    setName('');
    setPhone('');
    setDescription('');
    setPhotos([]);
    setAddress('');
  };

  return (
    <View style={styles.container}>
      <Text style={styles.title}>Post a Job</Text>
      <TextInput style={styles.input} placeholder="Name" value={name} onChangeText={setName} />
      <TextInput
        style={styles.input}
        placeholder="Phone"
        value={phone}
        onChangeText={setPhone}
        keyboardType="phone-pad"
      />
      <TextInput
        style={styles.input}
        placeholder="Job Description"
        value={description}
        onChangeText={setDescription}
        multiline
      />
      <TextInput style={styles.input} placeholder="Address" value={address} onChangeText={setAddress} />
      <Button title="Pick Photos (3–5)" onPress={pickPhotos} />
      <Text>{photos.length} photos selected</Text>
      <Button title="Post Job" onPress={postJob} />
    </View>
  );
};

// Customer: Review Screen
const CustomerReviewScreen = ({ navigation }) => {
  const [jobId, setJobId] = useState('');
  const [rating, setRating] = useState(1);
  const [comment, setComment] = useState('');

  const submitReview = async () => {
    if (!jobId) {
      Alert.alert('Error', 'Enter a job ID, you forgetful fuck!');
      return;
    }
    const review = { id: Date.now().toString(), jobId, rating, comment, timestamp: Date.now() };
    reviews.push(review);
    await AsyncStorage.setItem('reviews', JSON.stringify(reviews));
    Alert.alert('Success', 'Review submitted! Moving to job completion.');
    navigation.navigate('CustomerJobCompletion', { jobId });
    setJobId('');
    setRating(1);
    setComment('');
  };

  return (
    <View style={styles.container}>
      <Text style={styles.title}>Review Tradie</Text>
      <TextInput
        style={styles.input}
        placeholder="Job ID"
        value={jobId}
        onChangeText={setJobId}
      />
      <Text style={styles.label}>Star Rating</Text>
      <Picker
        selectedValue={rating}
        style={styles.picker}
        onValueChange={(itemValue) => setRating(itemValue)}
      >
        {[1, 2, 3, 4, 5].map((star) => (
          <Picker.Item key={star} label={`${star} Star${star > 1 ? 's' : ''}`} value={star} />
        ))}
      </Picker>
      <TextInput
        style={styles.input}
        placeholder="Comment (Optional)"
        value={comment}
        onChangeText={setComment}
        multiline
      />
      <Button title="Submit Review" onPress={submitReview} />
    </View>
  );
};

// Customer: Job Completion Screen
const CustomerJobCompletionScreen = ({ route }) => {
  const { jobId } = route.params || {};
  const [photos, setPhotos] = useState([]);
  const [satisfaction, setSatisfaction] = useState('Satisfied');

  const pickPhotos = () => {
    ImagePicker.launchImageLibrary({ selectionLimit: 5 }, (response) => {
      if (response.assets) {
        setPhotos(response.assets.map((asset) => asset.uri));
      }
    });
  };

  const submitCompletion = async () => {
    if (!jobId || photos.length < 1) {
      Alert.alert('Error', 'Select a job and upload at least 1 photo, you lazy fuck!');
      return;
    }
    const isCompliant = true; // Placeholder: AI vision API for Australian Building Code
    const completion = { jobId, photos, satisfaction, isCompliant, timestamp: Date.now() };
    reviews.push(completion);
    await AsyncStorage.setItem('reviews', JSON.stringify(reviews));
    if (satisfaction === 'Dissatisfied' || !isCompliant) {
      Alert.alert('Dispute', 'Admin notified for resolution!');
    } else {
      Alert.alert('Success', 'Job approved! Escrow payment released.');
    }
    setPhotos([]);
    setSatisfaction('Satisfied');
  };

  return (
    <View style={styles.container}>
      <Text style={styles.title}>Complete Job</Text>
      <TextInput
        style={styles.input}
        placeholder="Job ID"
        value={jobId || ''}
        editable={false}
      />
      <Button title="Pick Photos (up to 5)" onPress={pickPhotos} />
      <Text>{photos.length} photos selected</Text>
      <Picker
        selectedValue={satisfaction}
        style={styles.picker}
        onValueChange={(itemValue) => setSatisfaction(itemValue)}
      >
        <Picker.Item label="Satisfied" value="Satisfied" />
        <Picker.Item label="Dissatisfied" value="Dissatisfied" />
      </Picker>
      <Button title="Submit Completion" onPress={submitCompletion} />
    </View>
  );
};

// Tradie: Job Browsing Screen
const TradieJobScreen = () => (
  <View style={styles.container}>
    <Text style={styles.title}>Browse Jobs</Text>
    <FlatList
      data={jobs.filter((j) => j.status === 'Open')}
      keyExtractor={(item) => item.id}
      renderItem={({ item }) => (
        <View style={styles.listItem}>
          <Text>
            {item.name}: {item.description} ({item.address})
          </Text>
          <Button
            title="Submit Quote"
            onPress={() => Alert.alert('Quote', `Submit quote for ${item.id}`)}
          />
        </View>
      )}
    />
  </View>
);

// Tradie: Quoting Screen
const TradieQuoteScreen = () => {
  const [jobId, setJobId] = useState('');
  const [amount, setAmount] = useState('');

  const submitQuote = async () => {
    if (!jobId || !amount) {
      Alert.alert('Error', 'Fill out job ID and amount, you slacker!');
      return;
    }
    const quote = { id: Date.now().toString(), jobId, amount: parseFloat(amount), status: 'Pending' };
    quotes.push(quote);
    await AsyncStorage.setItem('quotes', JSON.stringify(quotes));
    Alert.alert('Success', 'Quote sent to customer!');
    setJobId('');
    setAmount('');
  };

  return (
    <View style={styles.container}>
      <Text style={styles.title}>Submit Quote</Text>
      <TextInput
        style={styles.input}
        placeholder="Job ID"
        value={jobId}
        onChangeText={setJobId}
      />
      <TextInput
        style={styles.input}
        placeholder="Quote Amount ($)"
        value={amount}
        onChangeText={setAmount}
        keyboardType="numeric"
      />
      <Button title="Submit Quote" onPress={submitQuote} />
    </View>
  );
};

// Customer Tab Navigator
const CustomerTabNavigator = () => (
  <Tab.Navigator>
    <Tab.Screen name="PostJob" component={CustomerJobScreen} />
    <Tab.Screen name="Review" component={CustomerReviewScreen} />
    <Tab.Screen
      name="CustomerJobCompletion"
      component={CustomerJobCompletionScreen}
      options={{ title: 'Complete Job' }}
    />
  </Tab.Navigator>
);

// Tradie Tab Navigator
const TradieTabNavigator = () => (
  <Tab.Navigator>
    <Tab.Screen name="BrowseJobs" component={TradieJobScreen} />
    <Tab.Screen name="SubmitQuote" component={TradieQuoteScreen} />
  </Tab.Navigator>
);

// Main App
const App = () => {
  return (
    <NavigationContainer>
      <Stack.Navigator initialRouteName="Home">
        <Stack.Screen name="Home" component={HomeScreen} />
        <Stack.Screen name="SignUp" component={SignUpScreen} />
        <Stack.Screen name="Login" component={LoginScreen} />
        <Stack.Screen name="CustomerTabs" component={CustomerTabNavigator} />
        <Stack.Screen name="TradieTabs" component={TradieTabNavigator} />
      </Stack.Navigator>
    </NavigationContainer>
  );
};
